== 2. list-set! ==
B1   (define L (list 1 2 3)) (list-set! L 5 'x) (display L)     => error: list-set!: index 5 is out of range for a list of length 3; valid indices are 0 to 2 5
B2   (list-set! (list) 0 'x) (display (null? '()))              => error: list-set!: index 0 is out of range — the list is empty, so no index is valid 0
B3   (define M (list 1 2 3)) (list-set! M 0 'ok) (display M)    => (ok 2 3)
B4   (list-set! (list 1 2) -1 'x) (display "no error")          => error: list-set!: index -1 is out of range for a list of length 2; valid indices are 0 to 1 -1
B5   (list-set! (list 1 2) 1.5 'x) (display "no error")         => error: list-set!: expected an integer index, got the number 1.5 1.5

== 3. assv / memv ==
C1   (display (memv 1/2 (list 1/3 1/2)))                        => (1/2)
C2   (display (assv 1/2 (list (cons 1/3 'a) (cons 1/2 'b))))    => (1/2 . b)
C3   (display (memv 2 (list 1 2 3)))                            => (2 3)
C4   (display (assv 'b (list (cons 'a 1) (cons 'b 2))))         => (b . 2)
C5   (display (memv (list 1) (list (list 1) 2)))                => #f
C6   (display (if (memv (make-rectangular 3 4) (list (make-rectangular 1 1) (make-rectangular 3 4))) "found" "#f")) => found
C7   (display (memv 1/2 (list 0.5)))                            => #f
C8   (display (if (memv #\b (list #\a #\b)) "found" "#f"))      => found
C9   (display (assv 4/2 (list (cons 2 'two))))                  => (2 . two)
C10  (display (assv 1/2 (list (list 1/2 'v))))                  => (1/2 v)
