;; The forces runtime — a tiny physics world built on real Motoi primitives.
;; Positions and velocities are vecs. Every world/step integrates forces with
;; forward Euler: velocity += acceleration, position += velocity. Gravity, wind,
;; drag, floor-bounce and point-attraction are ordinary vec arithmetic — the
;; same kinematics as phys/kinematics, unrolled one frame at a time.
(define PI (* 4 (atan 1)))
(define (dsqrt x) (sqrt x))
(define (dcos x) (cos x))
(define (dsin x) (sin x))
(define (datan2 y x) (atan y x))
(define *world* '())
(define *gravity* (vec/make 0 0))
(define *wind* (vec/make 0 0))
(define *floor* #f)          ; y-level of the floor, or #f
(define *attractors* '())    ; list of (center strength)
(define (world/reset!)
  (set! *world* '()) (set! *gravity* (vec/make 0 0))
  (set! *wind* (vec/make 0 0)) (set! *floor* #f) (set! *attractors* '()))
(define (world/gravity! gx gy) (set! *gravity* (vec/make gx gy)))
(define (world/wind! wx wy) (set! *wind* (vec/make wx wy)))
(define (world/floor! y) (set! *floor* y))
(define (world/attract! cx cy strength . falloff)
  ;; replace any attractor already at this center, so re-calling each frame
  ;; (to ease strength in) updates rather than accumulates.
  (set! *attractors*
        (cons (list (vec/make cx cy) strength)
              (filter (lambda (a) (not (and (= (car (car a)) cx)
                                            (= (cadr (car a)) cy))))
                      *attractors*))))
;; entity = vector: pos vel gscale drag bounce mass alive
(define (world/spawn tag x y . opts)
  (let ((e (vector (vec/make x y) (vec/make 0 0) 1.0 1.0 0.0 1.0 #t '())))
    (set! *world* (cons e *world*)) e))
(define (entity/set! e key val)
  (vector-set! e 7 (cons (cons key val)
                         (filter (lambda (p) (not (eq? (car p) key))) (vector-ref e 7)))))
(define (entity/get e key)
  (let ((p (assq key (vector-ref e 7)))) (if p (cdr p) #f)))
(define (entity/pos e) (vector-ref e 0))
(define (entity/vel e) (vector-ref e 1))
(define (entity/x e) (car (vector-ref e 0)))
(define (entity/y e) (cadr (vector-ref e 0)))
(define (entity/set-pos! e x y) (vector-set! e 0 (vec/make x y)))
(define (entity/set-vel! e vx vy) (vector-set! e 1 (vec/make vx vy)))
(define (entity/gravity-scale! e s) (vector-set! e 2 s))
(define (entity/drag! e d) (vector-set! e 3 d))
(define (entity/bounce! e b) (vector-set! e 4 b))
(define (entity/mass! e m) (vector-set! e 5 m))
(define (entity/impulse! e ix iy)
  (vector-set! e 1 (vec/add (vector-ref e 1)
                            (vec/scale (vec/make ix iy) (/ 1.0 (vector-ref e 5))))))
(define (world/impulse! ix iy)
  (for-each (lambda (e) (entity/impulse! e ix iy)) *world*))
(define (entity/distance a b)
  (let ((pa (entity/pos a)) (pb (entity/pos b)))
    (sqrt (+ (expt (- (car pa) (car pb)) 2) (expt (- (cadr pa) (cadr pb)) 2)))))
(define (world/count) (length *world*))
(define (world/each f) (for-each f *world*))
;; world/knowledge names the natural metaphor a force field simulates; it is a
;; pure annotation — it returns the (kind name) pair and has no effect on motion.
(define (world/knowledge kind name) (list kind name))
(define (step-one dt)
  (for-each
   (lambda (e)
     (let* ((pos (vector-ref e 0)) (vel (vector-ref e 1))
            (gs (vector-ref e 2)) (dr (vector-ref e 3)) (bo (vector-ref e 4))
            ;; acceleration = scaled gravity + wind + attractor pulls
            (acc (vec/add (vec/scale *gravity* gs) *wind*)))
       (for-each
        (lambda (a)
          (let* ((c (car a)) (s (cadr a))
                 (d (vec/sub c pos)) (len (sqrt (+ (expt (car d) 2) (expt (cadr d) 2)))))
            (when (> len 0.001)
              (set! acc (vec/add acc (vec/scale d (/ s (* len len))))))))
        *attractors*)
       (let* ((nv (vec/scale (vec/add vel acc) dr))
              (np (vec/add pos nv)))
         ;; floor collision + bounce
         (when (and *floor* (< (cadr np) *floor*))
           (set! np (vec/make (car np) *floor*))
           (set! nv (vec/make (car nv) (* (- (cadr nv)) bo))))
         (vector-set! e 1 nv)
         (vector-set! e 0 np))))
   *world*))
(define (world/step frames)
  (let loop ((k frames)) (when (> k 0) (step-one 1) (loop (- k 1)))))
