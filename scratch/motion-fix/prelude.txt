;; --- Motion runtime: a tiny world built on real Motoi primitives ---
;; Positions and velocities are vecs (plain (x y) lists, so car/cadr work).
;; Integration is forward Euler on vec/add + vec/scale. Nothing here is
;; special engine magic — it is ordinary Scheme over vec/*, sqrt, sin, cos, atan.
(define PI (* 4 (atan 1)))
(define (dsqrt x) (sqrt x))
(define (dcos x) (cos x))
(define (dsin x) (sin x))
(define (datan2 y x) (atan y x))
(define *world* '())
(define (world/reset!) (set! *world* '()))
(define (world/spawn tag x y)
  (let ((e (vector tag (vec/make x y) (vec/make 0 0))))
    (set! *world* (cons e *world*))
    e))
(define (entity/set-vel! e vx vy) (vector-set! e 2 (vec/make vx vy)))
(define (entity/set-pos! e x y)  (vector-set! e 1 (vec/make x y)))
(define (entity/vel e) (vector-ref e 2))
(define (entity/pos e) (vector-ref e 1))
(define (entity/x e)   (car  (vector-ref e 1)))
(define (entity/y e)   (cadr (vector-ref e 1)))
(define (world/step dt)
  (for-each
   (lambda (e) (vector-set! e 1 (vec/add (vector-ref e 1)
                                         (vec/scale (vector-ref e 2) dt))))
   *world*))
(define (world/count) (length *world*))
(define (entity/distance a b)
  (let ((pa (entity/pos a)) (pb (entity/pos b)))
    (sqrt (+ (expt (- (car pa) (car pb)) 2)
             (expt (- (cadr pa) (cadr pb)) 2)))))
(define (entity/overlaps? a b) (< (entity/distance a b) 0.5))
;; --- end runtime ---
