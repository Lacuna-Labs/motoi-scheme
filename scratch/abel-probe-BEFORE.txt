A2   (display (cdddr (list 1)))                                 => ()

A4   (display (cdar (list (cons 'a 1))))                        => 1

== 2. list-set! ==
B1   (define L (list 1 2 3)) (list-set! L 5 'x) (display L)     => (1 2 3   x)
B2   (list-set! (list) 0 'x) (display (null? '()))              => #t
B3   (define M (list 1 2 3)) (list-set! M 0 'ok) (display M)    => (ok 2 3)
B4   (list-set! (list 1 2) -1 'x) (display "no error")          => no error
B5   (list-set! (list 1 2) 1.5 'x) (display "no error")         => no error

== 3. assv / memv ==

C3   (display (memv 2 (list 1 2 3)))                            => (2 3)
C4   (display (assv 'b (list (cons 'a 1) (cons 'b 2))))         => (b . 2)
C5   (display (memv (list 1) (list (list 1) 2)))                => #f
C6   (display (if (memv (make-rectangular 3 4) (list (make-rectangular 1 1) (make-rectangular 3 4))) "found" "#f")) => #f
C7   (display (memv 1/2 (list 0.5)))                            => #f
C8   (display (if (memv #\b (list #\a #\b)) "found" "#f"))      => found
C9   (display (assv 4/2 (list (cons 2 'two))))                  => #f
C10  (display (assv 1/2 (list (list 1/2 'v))))                  => #f
